---
import { WebForms, Bind, BindForm, BInput, BButton } from '@astro-utils/forms/forms.js';

const bind = Bind({ name: '', sent: false });

function greet() { 
  bind.sent = true; 
}
---
<html lang="en">
  <head><title>Say hello</title></head>
  <body>
    <WebForms>
      <BindForm {bind}>
        <label for="name">Your name</label>
        <BInput id="name" name="name" required />

        <BButton onClick={greet} whenFormOK>Say hello</BButton>

        {bind.sent && <p>Hello, {bind.name}!</p>}
      </BindForm>
    </WebForms>
  </body>
</html>
